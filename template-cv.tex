    \cvpdftitle{$pdftitle$}
$endif$

% Document starts
\begin{document}

% use fancy on first page for correct footer formatting
\thispagestyle{fancy}

\cvfooter{$thisrepolink$}{$thisreponame$}


% Title and Contact
\begin{center}
    \begin{minipage}[t]{0.6\textwidth}
        \raggedright
        \cvname{$name$} \\
        \vspace{1.0cm} % tweak the spacing depending on how many contact items you have
        \cvtitle{$jobtitle$}
    \end{minipage}%
    %\vspace{0.5cm} % Add some space before sections
    \begin{minipage}[t]{0.4\textwidth}
        \raggedleft
        \vspace{-\baselineskip} % Pulls the first line up to match the name's baseline
        \cvcontact{\href{tel:$phone$}{%
            $phone$
            % \faPhone  % comment this line for no icon
        }} \\
        \cvcontact{\href{mailto:$email$}{%
            $email$
            % \faEnvelope % comment this line for no icon
        }} \\
        $if(website)$
            \cvcontact{\href{https://$website$}{%
                $website$
                % \faGlobe % comment this line for no icon
            }} \\
        $endif$
        $if(linkedin)$
            \cvcontact{\href{https://www.linkedin.com/in/$linkedin$}{%
                linkedin.com/in/$linkedin$
                % \faLinkedin % comment this line for no icon
            }} \\
        $endif$
        $if(github)$
            \cvcontact{\href{https://www.github.com/$github$}{%
                github.com/$github$
                % \faGithub  % comment this line for no icon
            }} \\
        $endif$
        $if(address)$
            \cvcontact{$address$} \\
        $endif$
    \end{minipage}
\end{center}

\vspace{0.5cm} % Add some space before sections
